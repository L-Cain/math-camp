\section{Problem Set Part I}
 
\begin{p} Let $L = a p^r (1 - p)^{n-r}$. Find $\log_{e}L$.
\end{p}
\begin{align*}
\log_{e}L&=\log_{e} a p^r (1 - p)^{n-r} \tag{by definition of $L$}\\
&=\log_{e} a +\log_{e} p^r +\log_{e} (1 - p)^{n-r} \tag{$\log(ab)=\log(a)+\log(b)$}\\
&=\log_{e} a +r\log_{e} p +(n-r)\log_{e} (1 - p) \tag{$\log(a^b)=b\log(a)$}
\end{align*}
