\begin{enumerate}
\item A composite function can be written $g(f(x))$ where $g$ and $f$ are basic functions.  Identify the basic functions in the following.
\begin{enumerate}
\item $h(x)=e^{-x^2}$, $g(x)=e^x$, $f(x)=-x^2$
\bigskip
\item $h(x)=(x^2+3x-2)^4$,  $g(x)=x^4$, $f(x)=x^2+3x-22$
\bigskip
\item $h(x)=sin(log(2*x+2))$, $g(x)=sin(x)$, $f(x)=log(x)$, $j(x)=2x+2$
\bigskip
\end{enumerate}
\item Find the derivative of the basic functions with respect to $x$ in each of the above cases.
\begin{enumerate}
\item $g(x)=e^{x}$, $g'(x)=e^x$, $f(x)=-x^2$, $f'(x)=-2x$
\bigskip
\item $g(x)=x^4$, $g'(x)=4x^3$, $f(x)=x^2+3x-22$, $f'(x)=2x+3$
\bigskip
\item $g(x)=sin(x)$, $g'(x)=cos(x)$, $f(x)=log(x)$, $f'(x)=1/x$   $j(x)=2x+2$,  $j'(x)=2$
\bigskip
\end{enumerate}
\item Find the derivative of $h(x)$ with respect to $x$ in each of the above cases.

\item $h(x)=e^{-x^2}$, $\frac{dh}{dx}=e^{-x^2}*(-2x)$
\bigskip
\item $h(x)=(x^2+3x-2)^4$, $\frac{dh}{dx}=4(x^2+3x-2)^3*(2x+3)$
\bigskip
\item $h(x)=sin(log(2*x+2))$, $\frac{dh}{dx}=cos(log(2*x+2))*\frac{1}{2*x+2}*2$
\bigskip
\end{enumerate}
